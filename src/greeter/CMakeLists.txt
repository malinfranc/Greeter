# To add a feature: create the .h in include/greeter/, the .cpp here,
# and add the .h and .cpp to the list below.
add_library(greeter STATIC)
target_sources(greeter
 PRIVATE #Definition des fichiers sources doit etre inaccecssible par user
 "${PROJECT_SOURCE_DIR}/src/greeter/message.cpp"
 "${PROJECT_SOURCE_DIR}/src/greeter/build_info.cpp"
 "${PROJECT_SOURCE_DIR}/include/greeter/message.h"
 "${PROJECT_SOURCE_DIR}/include/greeter/build_info.h"
)
target_include_directories(greeter
 PUBLIC #Definition de API, ce qui doit etre accessible 
 # Headers of the library, for the targets that use it
 "${PROJECT_SOURCE_DIR}/include"
 PRIVATE #Fichier config.h généré en intern par Cmake pas accessible !
 # greeter_config.h, generated by CMake in the build folder,
 # is only used inside the library
 "${PROJECT_BINARY_DIR}"
)
